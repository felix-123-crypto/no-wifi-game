    const difficulties = Array.from({ length: 100 }, (_, index) => {
      const level = index + 1;
      const pressure = index / 99;
      return {
        level,
        name: level === 1 ? "Extra Easy" : `Level ${level}`,
        scoreMultiplier: level,

    function currentDifficulty() {
      return difficulties[difficultyLevel - 1];
    }

    function rampProgress(now) {
      if (!runStartedAt) return 0;
      const elapsed = now - runStartedAt;
      if (elapsed <= beginnerGraceMs) return 0;
      return Math.min(1, (elapsed - beginnerGraceMs) / rampDurationMs);
    }

    function activeDifficulty(now) {
      const selected = currentDifficulty();
      const progress = rampProgress(now);
      return {
        ...selected,
        ghostDelay: Math.round(selected.ghostDelay + (760 - selected.ghostDelay) * (1 - progress)),
        chaseChance: selected.chaseChance * (0.18 + progress * 0.82)
      };
    }

    function elapsedMs(now = performance.now()) {
      if (gameState === "won" || gameState === "lost") return finishedElapsed;
      if ((gameState !== "playing" && gameState !== "dying") || !runStartedAt) return elapsedBeforeStart;
      return elapsedBeforeStart + now - runStartedAt;
    }

    function formatTime(ms) {
      const totalSeconds = Math.floor(ms / 1000);
      const minutes = Math.floor(totalSeconds / 60);
      const seconds = String(totalSeconds % 60).padStart(2, "0");
      return `${minutes}:${seconds}`;
    }

    function updateDifficultyControls() {
      const settings = currentDifficulty();
      difficultyLabel.textContent = `${settings.name} (${difficultyLevel} / 100) x${settings.scoreMultiplier} score`;
      difficultySlider.value = difficultyLevel;
      difficultyNumber.value = difficultyLevel;
    }

    function setDifficulty(level, shouldReset = true) {
      difficultyLevel = Math.max(1, Math.min(100, Number(level) || 1));
      updateDifficultyControls();
      if (shouldReset) resetGame(false);
    }

    function resetGame(startNow = true) {
      grid = map.map((row, y) => row.split("").map((cell, x) => {
        if (cell === "#") return "#";
        if (cell === "P") player = { x, y, dir: { x: 0, y: 0 }, mouth: 0 };
        if (cell === "P" || cell === "G") return " ";
        return cell === "C" ? "C" : ".";
      }));

      ghosts = [];
      map.forEach((row, y) => {
        [...row].forEach((cell, x) => {
          if (cell === "G") {
            ghosts.push({
              x,
              y,
              color: ghostColors[ghosts.length % ghostColors.length]
            });
          }
        });
      });

      score = 0;
      lives = 3;
      cherryUntil = 0;
      elapsedBeforeStart = 0;
      finishedElapsed = 0;
      wantedDir = { x: 0, y: 0 };
      lastPlayerMove = 0;
      lastGhostMove = 0;
      hitCooldownUntil = 0;
      deathStartedAt = 0;
      lastCherrySpawn = performance.now();
      runStartedAt = startNow ? performance.now() : 0;
      floatingTexts = [];
      hideToast();
      countPellets();
      updateHud();
      gameState = startNow ? "playing" : "ready";
      if (startNow) hideMessage();
      else showStartMessage();
    }

    function countPellets() {
      pelletsLeft = grid.flat().filter((cell) => cell === ".").length;
    }

    function updateHud() {
      scoreEl.textContent = score;
      livesEl.textContent = lives;
      pelletsEl.textContent = pelletsLeft;
      timerEl.textContent = formatTime(elapsedMs());
    }

    function flashStat(stat, kind) {
      if (!stat) return;
      const className = kind === "bad" ? "flash-bad" : "flash-good";
      stat.classList.remove("flash-good", "flash-bad");
      window.setTimeout(() => {
        stat.classList.add(className);
        window.setTimeout(() => stat.classList.remove(className), 280);
      }, 0);

    function showToast(text, kind = "good") {
      toast.textContent = text;
      toast.className = `toast show ${kind}`;
      window.clearTimeout(toastTimer);
      toastTimer = window.setTimeout(hideToast, 950);
    }

    function hideToast() {
      window.clearTimeout(toastTimer);
      toast.className = "toast";
      toast.textContent = "";

    function addFloatingText(text, x, y, color) {
      floatingTexts.push({

        y: y * tile + tile / 2,

    function playPointSound(kind = "pellet") {
      try {
        audioContext ||= new (window.AudioContext || window.webkitAudioContext)();
        const oscillator = audioContext.createOscillator();
        const gain = audioContext.createGain();
        const now = audioContext.currentTime;
        oscillator.type = "square";
        oscillator.frequency.setValueAtTime(kind === "ghost" ? 660 : 440, now);
        oscillator.frequency.exponentialRampToValueAtTime(kind === "ghost" ? 990 : 620, now + 0.08);
        gain.gain.setValueAtTime(0.0001, now);
        gain.gain.exponentialRampToValueAtTime(0.05, now + 0.01);
        gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.12);
        oscillator.connect(gain);
        gain.connect(audioContext.destination);
        oscillator.start(now);
        oscillator.stop(now + 0.13);
      } catch (_) {
        // Browsers may block audio until a user gesture; gameplay still works without sound.
      }
    }

    async function toggleFullscreen() {
      try {
        if (!document.fullscreenElement) {
          await document.querySelector(".game-wrap").requestFullscreen();
        } else {
          await document.exitFullscreen();

    function unlockAudio() {
      try {
        audioContext ||= new (window.AudioContext || window.webkitAudioContext)();
        if (audioContext.state === "suspended") {
          audioContext.resume();
        }
      } catch (_) {
        audioContext = null;
      }

    function canMove(x, y) {
      return x >= 0 && x < cols && y >= 0 && y < rows && grid[y][x] !== "#";
    }

    function sameDir(a, b) {
      return a.x === b.x && a.y === b.y;
    }

    function movePlayer(now) {
      if (now - lastPlayerMove < currentDifficulty().playerDelay) return;
      lastPlayerMove = now;

      const nextWanted = { x: player.x + wantedDir.x, y: player.y + wantedDir.y };
      if (canMove(nextWanted.x, nextWanted.y)) {
        player.dir = wantedDir;
      }

      const next = { x: player.x + player.dir.x, y: player.y + player.dir.y };
      if (canMove(next.x, next.y)) {
        player.x = next.x;
        player.y = next.y;
      }

      const cell = grid[player.y][player.x];
      if (cell === ".") {
        const points = 10 * currentDifficulty().scoreMultiplier;
        grid[player.y][player.x] = " ";
        score += points;
        pelletsLeft -= 1;
        playPointSound("pellet");
        flashStat(scoreStat, "good");
        addFloatingText(`+${points}`, player.x, player.y, "#55e6a5");
      } else if (cell === "C") {
        grid[player.y][player.x] = " ";
        cherryUntil = Math.max(cherryUntil, now) + currentDifficulty().cherryMs;
        showToast("+30 seconds cherry power", "good");
        addFloatingText("+30s", player.x, player.y, "#ff5b7c");
      }

      if (pelletsLeft === 0 && gameState === "playing") {
        finishedElapsed = elapsedMs(now);
        gameState = "won";
        showMessage("You Win!", `Final score: ${score}. Time: ${formatTime(finishedElapsed)}. Every pellet is gone.`, "Play Again");
      }
    }

    function moveGhosts(now) {
      const settings = activeDifficulty(now);
      if (now - lastGhostMove < settings.ghostDelay) return;
      lastGhostMove = now;

      ghosts.forEach((ghost) => {
        const options = dirs
          .map((dir) => ({ dir, x: ghost.x + dir.x, y: ghost.y + dir.y }))
          .filter((option) => canMove(option.x, option.y));

        if (!options.length) return;

        const powered = now < cherryUntil;
        const ranked = options.sort((a, b) => {
          const da = Math.abs(a.x - player.x) + Math.abs(a.y - player.y);
          const db = Math.abs(b.x - player.x) + Math.abs(b.y - player.y);
          return powered ? db - da : da - db;
        });

        const chase = Math.random() < settings.chaseChance;
        const choice = chase ? ranked[0] : ranked[Math.floor(Math.random() * ranked.length)];
        ghost.x = choice.x;
        ghost.y = choice.y;
      });
    }

    function handleCollisions(now) {
      if (gameState !== "playing") return;

      for (let index = ghosts.length - 1; index >= 0; index -= 1) {
        const ghost = ghosts[index];
        if (ghost.x !== player.x || ghost.y !== player.y) continue;

        if (now < cherryUntil) {
          const points = 30 * currentDifficulty().scoreMultiplier;
          score += points;
          ghosts.splice(index, 1);
          playPointSound("ghost");
          flashStat(scoreStat, "good");
          showToast(`Ghost cleared! +${points}`, "good");
          addFloatingText(`+${points}`, player.x, player.y, "#55e6a5");
          updateHud();
          continue;
        }

        if (now < hitCooldownUntil) continue;

        lives -= 1;
        gameState = "dying";
        deathStartedAt = now;
        flashStat(livesStat, "bad");
        showToast(`Life lost! ${Math.max(0, lives)} left`, "bad");
        addFloatingText("-1 life", player.x, player.y, "#ff5b7c");
        player.dir = { x: 0, y: 0 };
        wantedDir = { x: 0, y: 0 };
        updateHud();
        return;
      }
    }

    function finishDeath(now) {
      if (gameState !== "dying" || now - deathStartedAt < deathDuration) return;

      if (lives <= 0) {
        finishedElapsed = elapsedMs(now);
        gameState = "lost";
        showMessage("Game Over", `Final score: ${score}. Time: ${formatTime(finishedElapsed)}. Restart and try another route.`, "Try Again");
        updateHud();
        return;
      }

      player.x = 1;
      player.y = 1;
      player.dir = { x: 0, y: 0 };
      wantedDir = { x: 0, y: 0 };
      hitCooldownUntil = now + 1300;
      deathStartedAt = 0;
      gameState = "playing";
      updateHud();
    }

    function countCherries() {
      return grid.flat().filter((cell) => cell === "C").length;
    }

    function isOccupied(x, y) {
      return (player.x === x && player.y === y) || ghosts.some((ghost) => ghost.x === x && ghost.y === y);
    }

    function spawnCherry() {
      const openTiles = [];
      for (let y = 0; y < rows; y += 1) {
        for (let x = 0; x < cols; x += 1) {
          if (grid[y][x] === " " && !isOccupied(x, y)) {
            openTiles.push({ x, y });
          }
        }
      }

      if (!openTiles.length) return false;
      const choice = openTiles[Math.floor(Math.random() * openTiles.length)];
      grid[choice.y][choice.x] = "C";
      return true;
    }

    function updateCherrySpawns(now) {
      if (now - lastCherrySpawn < cherrySpawnInterval) return;
      lastCherrySpawn = now;

      if (countCherries() < maxCherries) {
        spawnCherry();
      }
    }

    function drawBoard(now) {
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      for (let y = 0; y < rows; y += 1) {
        for (let x = 0; x < cols; x += 1) {
          const px = x * tile;
          const py = y * tile;
          ctx.fillStyle = grid[y][x] === "#" ? wall : path;
          ctx.fillRect(px, py, tile, tile);

          if (grid[y][x] === "#") {
            ctx.fillStyle = "rgba(71, 200, 255, 0.28)";
            ctx.fillRect(px + 3, py + 3, tile - 6, 2);
            ctx.fillStyle = "rgba(13, 18, 48, 0.42)";
            ctx.fillRect(px + 3, py + tile - 5, tile - 6, 2);
          } else {
            ctx.fillStyle = "rgba(255, 255, 255, 0.025)";
            ctx.fillRect(px, py, tile, 1);
            ctx.fillRect(px, py, 1, tile);
          }

          if (grid[y][x] === ".") {
            ctx.fillStyle = "#fff3a6";
            ctx.beginPath();
            ctx.arc(px + tile / 2, py + tile / 2, 4, 0, Math.PI * 2);
            ctx.fill();
            ctx.fillStyle = "rgba(255, 228, 92, 0.28)";
            ctx.beginPath();
            ctx.arc(px + tile / 2, py + tile / 2, 7, 0, Math.PI * 2);
            ctx.fill();
          }

          if (grid[y][x] === "C") {
            ctx.fillStyle = "rgba(255, 79, 139, 0.2)";
            ctx.beginPath();
            ctx.arc(px + tile / 2, py + tile / 2, 13, 0, Math.PI * 2);
            ctx.fill();
            ctx.fillStyle = "#ff4f8b";
            ctx.beginPath();
            ctx.arc(px + tile / 2, py + tile / 2, 9, 0, Math.PI * 2);
            ctx.fill();
            ctx.fillStyle = "#ff9dbb";
            ctx.beginPath();
            ctx.arc(px + tile / 2 - 3, py + tile / 2 - 3, 3, 0, Math.PI * 2);
            ctx.fill();
            ctx.fillStyle = "#3ff0b5";
            ctx.fillRect(px + 16, py + 6, 3, 9);
          }
        }
      }

      drawGhosts(now);
      drawPlayer(now);
      drawFloatingTexts(now);

      if (now < cherryUntil) {
        const settings = currentDifficulty();
        const seconds = Math.max(0, Math.ceil((cherryUntil - now) / 1000));
        const barWidth = 184;
        const fillWidth = Math.min(barWidth, ((cherryUntil - now) / settings.cherryMs) * barWidth);
        ctx.fillStyle = "rgba(255, 91, 124, 0.92)";
        ctx.fillRect(12, canvas.height - 42, Math.max(0, fillWidth), 10);
        ctx.fillStyle = "#f5f7fb";
        ctx.font = "700 16px system-ui";
        ctx.fillText(`Cherry power: ${seconds}s`, 12, canvas.height - 52);
      }

      if (gameState === "playing") {
        const elapsed = Math.max(0, now - runStartedAt);
        const ramp = rampProgress(now);
        const label = elapsed < beginnerGraceMs
          ? `Warm-up: ${Math.ceil((beginnerGraceMs - elapsed) / 1000)}s`
          : `Danger ramp: ${Math.round(ramp * 100)}%`;
        ctx.fillStyle = "rgba(10, 13, 19, 0.72)";
        ctx.fillRect(canvas.width - 188, 12, 176, 28);
        ctx.fillStyle = "#f5f7fb";
        ctx.font = "800 15px system-ui";
        ctx.textAlign = "right";
        ctx.fillText(label, canvas.width - 22, 31);
        ctx.textAlign = "left";
      }
    }

    function drawFloatingTexts(now) {
      floatingTexts = floatingTexts.filter((item) => now - item.born < item.lifetime);
      floatingTexts.forEach((item) => {
        const age = now - item.born;
        const progress = age / item.lifetime;
        ctx.save();
        ctx.globalAlpha = 1 - progress;
        ctx.fillStyle = item.color;
        ctx.strokeStyle = "rgba(10, 13, 19, 0.86)";
        ctx.lineWidth = 4;
        ctx.font = "900 18px system-ui";
        ctx.textAlign = "center";
        ctx.strokeText(item.text, item.x, item.y - progress * 32);
        ctx.fillText(item.text, item.x, item.y - progress * 32);
        ctx.restore();
      });
    }

    function drawPlayer(now) {
      const cx = player.x * tile + tile / 2;
      const cy = player.y * tile + tile / 2;
      const mouth = Math.sin(now / 90) * 0.18 + 0.28;
      const angle = getAngle(player.dir);

      ctx.save();
      ctx.translate(cx, cy);
      ctx.rotate(angle);
      ctx.shadowColor = "rgba(255, 228, 92, 0.55)";
      ctx.shadowBlur = 14;

      if (gameState === "dying") {
        const progress = Math.min(1, (now - deathStartedAt) / deathDuration);
        const open = 0.2 + progress * 2.85;
        const radius = 12 * (1 - Math.max(0, progress - 0.72) / 0.28);
        ctx.globalAlpha = 1 - Math.max(0, progress - 0.78) / 0.22;
        ctx.fillStyle = "#ffe45c";
        ctx.beginPath();
        ctx.moveTo(0, 0);
        ctx.arc(0, 0, Math.max(0, radius), open, Math.PI * 2 - open);
        ctx.closePath();
        ctx.fill();
        ctx.restore();
        return;
      }

      ctx.fillStyle = now < hitCooldownUntil && Math.floor(now / 120) % 2 ? "#fff7b8" : "#ffe45c";
      ctx.beginPath();
      ctx.moveTo(0, 0);
      ctx.arc(0, 0, 12, mouth, Math.PI * 2 - mouth);
      ctx.closePath();
      ctx.fill();
      ctx.shadowBlur = 0;
      ctx.strokeStyle = "#2a1900";
      ctx.lineWidth = 2;
      ctx.stroke();
      ctx.fillStyle = "#2a1900";
      ctx.beginPath();
      ctx.arc(3, -6, 2.4, 0, Math.PI * 2);
      ctx.fill();
      ctx.restore();
    }

    function getAngle(dir) {
      if (sameDir(dir, { x: -1, y: 0 })) return Math.PI;
      if (sameDir(dir, { x: 0, y: -1 })) return -Math.PI / 2;
      if (sameDir(dir, { x: 0, y: 1 })) return Math.PI / 2;
      return 0;
    }

    function drawGhosts(now) {
      ghosts.forEach((ghost) => {
        const cx = ghost.x * tile + tile / 2;
        const cy = ghost.y * tile + tile / 2;
        const powered = now < cherryUntil;

        ctx.save();
        ctx.shadowColor = powered ? "rgba(63, 240, 181, 0.55)" : `${ghost.color}88`;
        ctx.shadowBlur = 10;
        ctx.fillStyle = powered ? "#55e6a5" : ghost.color;
        ctx.beginPath();
        ctx.arc(cx, cy - 2, 12, Math.PI, 0);
        ctx.lineTo(cx + 12, cy + 13);
        ctx.lineTo(cx + 6, cy + 9);
        ctx.lineTo(cx, cy + 13);
        ctx.lineTo(cx - 6, cy + 9);
        ctx.lineTo(cx - 12, cy + 13);
        ctx.closePath();
        ctx.fill();

        ctx.fillStyle = "#0a0d13";
        ctx.beginPath();
        ctx.arc(cx - 5, cy - 3, 2.8, 0, Math.PI * 2);
        ctx.arc(cx + 5, cy - 3, 2.8, 0, Math.PI * 2);
        ctx.fill();
        ctx.restore();
      });
    }

    function loop(now) {
      const elapsed = now - lastFrame;
      lastFrame = now;

      if (gameState === "playing" && elapsed >= 0) {
        updateCherrySpawns(now);
        movePlayer(now);
        moveGhosts(now);
        handleCollisions(now);
        updateHud();
      } else if (gameState === "dying") {
        finishDeath(now);
        updateHud();
      }

      drawBoard(now);
      requestAnimationFrame(loop);
    }
